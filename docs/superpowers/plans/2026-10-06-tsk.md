# tsk Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `tsk`, a terminal tool that dispatches Beads tasks to OpenCode agents in herdr tabs and shows every dispatched task's live state in one board.

**Architecture:** One Go binary with two Bubble Tea views (`tsk board`, `tsk project`). Small adapter packages wrap `bd`, the OpenCode v2 HTTP API, `herdr` and `git`. A pure `status` package derives each task's state, and `dispatch` runs the seven dispatch steps with undo. There is no background process: views load state when they open and follow OpenCode's event stream while open.

**Tech Stack:** Go (toolchain 1.27.1), `charm.land/bubbletea/v2` v2.0.10, `github.com/BurntSushi/toml` v1.6.0, and the standard library for HTTP, SSE and process execution.

**Spec:** `docs/superpowers/specs/2026-10-06-tsk-design.md`

## Global Constraints

- Module path: `github.com/AntoineGS/taskestrator`. The binary is built from `./cmd/tsk`.
- Only two third-party modules: `charm.land/bubbletea/v2@v2.0.10` and `github.com/BurntSushi/toml@v1.6.0`.
- Beads metadata keys are exactly `tsk.session`, `tsk.worktree` and `tsk.branch`. The finished status is `review`, in category `wip` (`status.custom` entry `review:wip`).
- Worktree path `<repo>/.worktrees/<task-id>`, branch `task/<task-id>`.
- Only the `tracker` package runs `bd`. Only `opencode` talks HTTP to OpenCode. Only `herdr` runs `herdr`. Only `gitwt` runs `git`.
- The OpenCode password is only ever placed in the `Authorization` header. It never appears in errors, logs, Beads or test output.
- tsk never pushes, merges or rebases. The agent prompt forbids push and merge.
- Live state is never written to Beads. Beads is written only at dispatch (claim and metadata), on dispatch undo, and when adding the `review` status.
- `make lint` (`go vet` plus `gofmt -l`) and `make test` pass at the end of every task.

## Review Focus

- A question or permission raised by a **subagent** (a child session) must mark the task as waiting on you. Tests: Task 6 `TestSnapshotCountsChildSessions`, Task 12 `TestBoardChildEventRefreshesRootSession`.
- If the OpenCode service **restarts** while the board is open, no stale state may remain. Expect a banner, a reconnect with backoff, then a full reload. Test: Task 12 `TestBoardReconnectsAndReloads`.
- One repo with a **malformed `.beads/metadata.json`** or an unreadable directory must not hide the other projects. Test: Task 3 `TestScanBadMetadataWarnsAndContinues`.
- **Re-dispatching** a lost task, or dispatching when `.worktrees/<id>` already exists, must reuse the worktree. An undo must never delete a worktree it didn't create. Tests: Task 9 `TestAddWorktreeReusesExisting`, Task 10 `TestRunDoesNotRemoveReusedWorktree`.
- A task **claimed outside tsk** (`in_progress` with no `tsk.session`) must not show as queued or be dispatchable. Tests: Task 4 case `untracked`, Task 13 `TestProjectRefusesUntrackedDispatch`.

// Package execx runs external commands for the adapter packages.
package execx

import (
	"bytes"
	"context"
	"errors"
	"fmt"
	"os/exec"
	"strings"
)

// Runner runs a command in dir (the current directory when empty) and
// returns its standard output.
type Runner func(ctx context.Context, dir string, args ...string) ([]byte, error)

// Command returns a Runner for bin.
func Command(bin string) Runner {
	return func(ctx context.Context, dir string, args ...string) ([]byte, error) {
		cmd := exec.CommandContext(ctx, bin, args...)
		cmd.Dir = dir
		var stderr bytes.Buffer
		cmd.Stderr = &stderr
		out, err := cmd.Output()
		if err != nil {
			return out, &Error{
				Cmd:    strings.TrimSpace(bin + " " + strings.Join(args, " ")),
				Stderr: strings.TrimSpace(stderr.String()),
				Err:    err,
			}
		}
		return out, nil
	}
}

// Error describes a failed command.
type Error struct {
	Cmd    string
	Stderr string
	Err    error
}

func (e *Error) Error() string {
	if e.Stderr != "" {
		return fmt.Sprintf("%s: %v: %s", e.Cmd, e.Err, e.Stderr)
	}
	return fmt.Sprintf("%s: %v", e.Cmd, e.Err)
}

func (e *Error) Unwrap() error { return e.Err }

// ExitCode returns the exit code carried by err, or -1 when there is none.
func ExitCode(err error) int {
	var ee *exec.ExitError
	if errors.As(err, &ee) {
		return ee.ExitCode()
	}
	return -1
}
```

// Package config loads tsk's user configuration and per-repository files.
package config

import (
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strings"
	"time"

	"github.com/BurntSushi/toml"
)

// Config is the user configuration from ~/.config/tsk/config.toml.
type Config struct {
	Roots          []string
	Hidden         []string
	RereadInterval time.Duration
}

type fileConfig struct {
	Roots          []string `toml:"roots"`
	Hidden         []string `toml:"hidden"`
	RereadInterval string   `toml:"reread_interval"`
}

// Default returns the configuration used when no file exists.
func Default() Config {
	return Config{Roots: []string{ExpandHome("~/gits")}, RereadInterval: 10 * time.Second}
}

// DefaultPath returns the user configuration file path.
func DefaultPath() string {
	if dir := os.Getenv("XDG_CONFIG_HOME"); dir != "" {
		return filepath.Join(dir, "tsk", "config.toml")
	}
	return ExpandHome("~/.config/tsk/config.toml")
}

// Load reads the configuration at path, falling back to defaults when it does not exist.
func Load(path string) (Config, error) {
	cfg := Default()
	var fc fileConfig
	_, err := toml.DecodeFile(path, &fc)
	if errors.Is(err, fs.ErrNotExist) {
		return cfg, nil
	}
	if err != nil {
		return Config{}, fmt.Errorf("read %s: %w", path, err)
	}
	if len(fc.Roots) > 0 {
		cfg.Roots = nil
		for _, r := range fc.Roots {
			cfg.Roots = append(cfg.Roots, ExpandHome(r))
		}
	}
	cfg.Hidden = fc.Hidden
	if fc.RereadInterval != "" {
		d, err := time.ParseDuration(fc.RereadInterval)
		if err != nil || d <= 0 {
			return Config{}, fmt.Errorf("%s: reread_interval %q is not a positive duration", path, fc.RereadInterval)
		}
		cfg.RereadInterval = d
	}
	return cfg, nil
}

// ProjectFile is a repository's optional .tsk.toml.
type ProjectFile struct {
	Setup  string `toml:"setup"`
	Prompt string `toml:"prompt"`
}

// LoadProjectFile reads repo/.tsk.toml, returning a zero value when it does not exist.
func LoadProjectFile(repo string) (ProjectFile, error) {
	var pf ProjectFile
	path := filepath.Join(repo, ".tsk.toml")
	_, err := toml.DecodeFile(path, &pf)
	if errors.Is(err, fs.ErrNotExist) {
		return ProjectFile{}, nil
	}
	if err != nil {
		return ProjectFile{}, fmt.Errorf("read %s: %w", path, err)
	}
	return pf, nil
}

// ExpandHome replaces a leading ~ with the user's home directory.
func ExpandHome(p string) string {
	if p != "~" && !strings.HasPrefix(p, "~/") {
		return p
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return p
	}
	return filepath.Join(home, strings.TrimPrefix(p, "~"))
}
```

// Package discover finds Beads projects under the configured roots.
package discover

import (
	"encoding/json"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

// Project is a repository with a Beads database.
type Project struct {
	Name string // Beads database name, e.g. "se"
	Repo string // absolute repository path, symlinks resolved
}

const maxDepth = 4

var skipDirs = map[string]bool{"node_modules": true, "vendor": true}

// Scan walks each root up to four levels deep and returns the projects found,
// sorted by name, plus warnings for anything it could not read.
func Scan(roots, hidden []string) ([]Project, []string) {
	hide := map[string]bool{}
	for _, h := range hidden {
		hide[h] = true
	}
	seen := map[string]bool{}
	var out []Project
	var warns []string
	for _, root := range roots {
		root = filepath.Clean(root)
		err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
			if err != nil {
				if path == root {
					return err
				}
				warns = append(warns, err.Error())
				return nil
			}
			if !d.IsDir() {
				return nil
			}
			if path != root {
				name := d.Name()
				if skipDirs[name] || strings.HasPrefix(name, ".") || depth(root, path) > maxDepth {
					return fs.SkipDir
				}
			}
			p, ok, err := readProject(path)
			if err != nil {
				warns = append(warns, err.Error())
				return nil
			}
			if ok && !hide[p.Name] && !hide[p.Repo] && !seen[p.Repo] {
				seen[p.Repo] = true
				out = append(out, p)
			}
			return nil
		})
		if err != nil {
			warns = append(warns, fmt.Sprintf("scan %s: %v", root, err))
		}
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Name < out[j].Name })
	return out, warns
}

func depth(root, path string) int {
	rel, err := filepath.Rel(root, path)
	if err != nil {
		return maxDepth + 1
	}
	return len(strings.Split(rel, string(filepath.Separator)))
}

func readProject(dir string) (Project, bool, error) {
	path := filepath.Join(dir, ".beads", "metadata.json")
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return Project{}, false, nil
	}
	if err != nil {
		return Project{}, false, err
	}
	var meta struct {
		Database string `json:"dolt_database"`
	}
	if err := json.Unmarshal(data, &meta); err != nil {
		return Project{}, false, fmt.Errorf("%s: %v", path, err)
	}
	if meta.Database == "" {
		return Project{}, false, nil
	}
	return Project{Name: meta.Database, Repo: resolve(dir)}, true, nil
}

func resolve(p string) string {
	if r, err := filepath.EvalSymlinks(p); err == nil {
		return r
	}
	return filepath.Clean(p)
}

// ForDir returns the project whose repository contains dir, preferring the deepest match.
func ForDir(projects []Project, dir string) (Project, bool) {
	dir = resolve(dir)
	var best Project
	found := false
	for _, p := range projects {
		if dir == p.Repo || strings.HasPrefix(dir, p.Repo+string(filepath.Separator)) {
			if !found || len(p.Repo) > len(best.Repo) {
				best, found = p, true
			}
		}
	}
	return best, found
}

// Package status derives a task's display state from its Beads status and
// the live state of its OpenCode session.
package status

// State is what the views show for a task.
type State int

const (
	Queued State = iota
	InProgress
	Waiting
	Review
	Done
	Lost
	Unknown
	Untracked
)

var names = [...]string{"queued", "in progress", "waiting on you", "review", "done", "lost", "unknown", "untracked"}

func (s State) String() string {
	if s >= 0 && int(s) < len(names) {
		return names[s]
	}
	return "invalid"
}

// Snapshot is the live state of a task's session tree.
type Snapshot struct {
	Exists      bool
	Running     bool
	Forms       int
	Permissions int
}

// Display is a derived state plus an optional detail, such as why a task is waiting.
type Display struct {
	State  State
	Detail string
}

func (d Display) String() string {
	if d.Detail == "" {
		return d.State.String()
	}
	return d.State.String() + " (" + d.Detail + ")"
}

// Derive applies the spec's status rules in order. snap is nil when the
// session's live state is unavailable.
func Derive(taskStatus, sessionID string, snap *Snapshot) Display {
	switch {
	case taskStatus == "closed":
		return Display{State: Done}
	case taskStatus == "review":
		return Display{State: Review}
	case sessionID == "" && taskStatus == "open":
		return Display{State: Queued}
	case sessionID == "":
		return Display{State: Untracked, Detail: taskStatus}
	case snap == nil:
		return Display{State: Unknown}
	case !snap.Exists:
		return Display{State: Lost}
	case snap.Forms > 0:
		return Display{State: Waiting, Detail: "question"}
	case snap.Permissions > 0:
		return Display{State: Waiting, Detail: "permission"}
	case snap.Running:
		return Display{State: InProgress}
	default:
		return Display{State: Waiting, Detail: "idle"}
	}
}

package tracker

import (

	"github.com/AntoineGS/taskestrator/internal/execx"

const (
	MetaSession  = "tsk.session"
	MetaWorktree = "tsk.worktree"
	MetaBranch   = "tsk.branch"

// Task is a Beads issue as returned by bd --json.
type Task struct {
	ID          string         `json:"id"`
	Title       string         `json:"title"`
	Description string         `json:"description"`
	Acceptance  string         `json:"acceptance_criteria"`
	Status      string         `json:"status"`
	Priority    int            `json:"priority"`
	Assignee    string         `json:"assignee"`
	Metadata    map[string]any `json:"metadata"`
	UpdatedAt   time.Time      `json:"updated_at"`
	ClosedAt    *time.Time     `json:"closed_at"`
}

// Meta returns a string metadata value, or "" when absent or not a string.
func (t Task) Meta(key string) string {
	if v, ok := t.Metadata[key].(string); ok {
		return v
	}
	return ""
}

// Tracker reads and updates a project's tasks.
type Tracker interface {
	Ready(ctx context.Context, repo string) ([]Task, error)
	List(ctx context.Context, repo string) ([]Task, error)
	Claim(ctx context.Context, repo, id string) error
	Unclaim(ctx context.Context, repo, id string) error
	SetMetadata(ctx context.Context, repo, id string, kv map[string]string) error
	UnsetMetadata(ctx context.Context, repo, id string, keys ...string) error
	HasReviewStatus(ctx context.Context, repo string) (bool, error)
	AddReviewStatus(ctx context.Context, repo string) error
}

// BD implements Tracker with the bd CLI, run inside the repository.
type BD struct{ Run execx.Runner }

// New returns a BD that runs the bd binary on PATH.
func New() BD { return BD{Run: execx.Command("bd")} }

func (b BD) Ready(ctx context.Context, repo string) ([]Task, error) {
	return b.tasks(ctx, repo, "ready", "--json", "-n", "0")
}

func (b BD) List(ctx context.Context, repo string) ([]Task, error) {
	return b.tasks(ctx, repo, "list", "--json", "--all", "-n", "0")
}

func (b BD) tasks(ctx context.Context, repo string, args ...string) ([]Task, error) {
	out, err := b.Run(ctx, repo, args...)
	if err != nil {
		return nil, err
	}
	var tasks []Task
	if err := json.Unmarshal(out, &tasks); err != nil {
		return nil, fmt.Errorf("bd %s: unexpected output: %w", args[0], err)
	}
	return tasks, nil
}

func (b BD) Claim(ctx context.Context, repo, id string) error {
	_, err := b.Run(ctx, repo, "update", id, "--claim")
	return err
}

func (b BD) Unclaim(ctx context.Context, repo, id string) error {
	_, err := b.Run(ctx, repo, "update", id, "--status", "open", "--assignee", "")
	return err
}

func (b BD) SetMetadata(ctx context.Context, repo, id string, kv map[string]string) error {
	keys := make([]string, 0, len(kv))
	for k := range kv {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	args := []string{"update", id}
	for _, k := range keys {
		args = append(args, "--set-metadata", k+"="+kv[k])
	}
	_, err := b.Run(ctx, repo, args...)
	return err
}

func (b BD) UnsetMetadata(ctx context.Context, repo, id string, keys ...string) error {
	args := []string{"update", id}
	for _, k := range keys {
		args = append(args, "--unset-metadata", k)
	}
	_, err := b.Run(ctx, repo, args...)
	return err
}

func (b BD) customStatuses(ctx context.Context, repo string) (string, error) {
	out, err := b.Run(ctx, repo, "config", "get", "status.custom", "--json")
	if err != nil {
		return "", err
	}
	var v struct {
		Value string `json:"value"`
	}
	if err := json.Unmarshal(out, &v); err != nil {
		return "", fmt.Errorf("bd config get status.custom: unexpected output: %w", err)
	}
	return strings.TrimSpace(v.Value), nil
}

func hasReview(custom string) bool {
	for _, entry := range strings.Split(custom, ",") {
		name, _, _ := strings.Cut(strings.TrimSpace(entry), ":")
		if name == StatusReview {
			return true
		}
	}
	return false
}

func (b BD) HasReviewStatus(ctx context.Context, repo string) (bool, error) {
	custom, err := b.customStatuses(ctx, repo)
	if err != nil {
		return false, err
	}
	return hasReview(custom), nil
}

func (b BD) AddReviewStatus(ctx context.Context, repo string) error {
	custom, err := b.customStatuses(ctx, repo)
	if err != nil {
		return err
	}
	if hasReview(custom) {
		return nil
	}
	value := StatusReview + ":wip"
	if custom != "" {
		value = custom + "," + value
	}
	_, err = b.Run(ctx, repo, "config", "set", "status.custom", value)
	return err
}

	BaseURL  string

		Directory string `json:"directory"`

	Title     string
	Directory string
	Metadata  map[string]string
}

func (c *Client) httpClient() *http.Client {
	if c.HTTP != nil {
		return c.HTTP
	}
	return http.DefaultClient
}

func (c *Client) newRequest(ctx context.Context, method, path string, body io.Reader) (*http.Request, error) {
	req, err := http.NewRequestWithContext(ctx, method, c.BaseURL+path, body)
	if err != nil {
		return nil, err
	}
	if c.Password != "" {
		req.SetBasicAuth("opencode", c.Password)
	}
	return req, nil
}

func (c *Client) do(ctx context.Context, method, path string, body, out any) error {
	var r io.Reader
	if body != nil {
		b, err := json.Marshal(body)
		if err != nil {
			return err
		}
		r = bytes.NewReader(b)
	}
	req, err := c.newRequest(ctx, method, path, r)
	if err != nil {
		return err
	}
	if body != nil {
		req.Header.Set("Content-Type", "application/json")
	}
	resp, err := c.httpClient().Do(req)
	if err != nil {
		return fmt.Errorf("%s %s: %w", method, path, err)
	}
	defer resp.Body.Close()
	if resp.StatusCode == http.StatusNotFound {
		return fmt.Errorf("%s %s: %w", method, path, ErrNotFound)
	}
	if resp.StatusCode >= 300 {
		msg, _ := io.ReadAll(io.LimitReader(resp.Body, 512))
		return fmt.Errorf("%s %s: HTTP %d: %s", method, path, resp.StatusCode, bytes.TrimSpace(msg))
	}
	if out == nil {
		return nil
	}
	return json.NewDecoder(resp.Body).Decode(out)
}

package opencode

import (
	"context"
	"errors"
	"fmt"

	"github.com/AntoineGS/taskestrator/internal/status"
)

const maxTree = 64

// Snapshot loads the live state of a task's session and its subagent
// sessions. active is the result of Active, shared across snapshots.
func (c *Client) Snapshot(ctx context.Context, id string, active map[string]bool) (status.Snapshot, error) {
	if _, err := c.GetSession(ctx, id); err != nil {
		if errors.Is(err, ErrNotFound) {
			return status.Snapshot{}, nil
		}
		return status.Snapshot{}, err
	}
	snap := status.Snapshot{Exists: true}
	seen := map[string]bool{}
	queue := []string{id}
	for len(queue) > 0 && len(seen) < maxTree {
		cur := queue[0]
		queue = queue[1:]
		if seen[cur] {
			continue
		}
		seen[cur] = true
		if active[cur] {
			snap.Running = true
		}
		forms, err := c.FormCount(ctx, cur)
		if err != nil && !errors.Is(err, ErrNotFound) {
			return status.Snapshot{}, err
		}
		snap.Forms += forms
		perms, err := c.PermissionCount(ctx, cur)
		if err != nil && !errors.Is(err, ErrNotFound) {
			return status.Snapshot{}, err
		}
		snap.Permissions += perms
		kids, err := c.Children(ctx, cur)
		if err != nil && !errors.Is(err, ErrNotFound) {
			return status.Snapshot{}, err
		}
		for _, k := range kids {
			queue = append(queue, k.ID)
		}
	}
	return snap, nil
}

// Root follows parent links from id to its top-level session. Parent links
// never change, so they are cached for the client's lifetime.
func (c *Client) Root(ctx context.Context, id string) (string, error) {
	for range maxTree {
		c.mu.Lock()
		parent, ok := c.parents[id]
		c.mu.Unlock()
		if !ok {
			s, err := c.GetSession(ctx, id)
			if err != nil {
				return "", err
			}
			parent = s.ParentID
			c.mu.Lock()
			if c.parents == nil {
				c.parents = map[string]string{}
			}
			c.parents[id] = parent
			c.mu.Unlock()
		}
		if parent == "" {
			return id, nil
		}
		id = parent
	}
	return "", fmt.Errorf("session %s: parent chain longer than %d", id, maxTree)
}

func FromService(ctx context.Context) (*Client, error) {
	out, err := exec.CommandContext(ctx, "opencode", "service", "status").Output()
	if err != nil {
		return nil, fmt.Errorf("opencode service status: %w (start it with: opencode service start)", err)
	}

	}
	dir, err := os.UserConfigDir()
	if err != nil {
		return nil, err
	}

	if err != nil {
		return nil, err
	}
	return &Client{BaseURL: base, Password: pw}, nil
}

func parseServiceURL(out string) string {
	for _, line := range strings.Split(out, "\n") {
		line = strings.TrimSpace(line)
		if strings.HasPrefix(line, "http://") || strings.HasPrefix(line, "https://") {
			return strings.TrimRight(line, "/")
		}
	}
	return ""
}

func readPassword(path string) (string, error) {
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return "", nil
	}
	if err != nil {
		return "", fmt.Errorf("read OpenCode service config: %w", err)
	}
	var cfg struct {
		Password string `json:"password"`
	}
	if err := json.Unmarshal(data, &cfg); err != nil {
		return "", fmt.Errorf("parse %s: %v", path, err)
	}
	return cfg.Password, nil

package opencode

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"
)

// Event is the part of an OpenCode event tsk uses.
type Event struct {
	Type      string
	SessionID string
}

// Subscribe streams events from the service. The channel closes when the
// connection ends or ctx is cancelled; callers reconnect.
func (c *Client) Subscribe(ctx context.Context) (<-chan Event, error) {
	req, err := c.newRequest(ctx, http.MethodGet, "/api/event", nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Accept", "text/event-stream")
	resp, err := c.httpClient().Do(req)
	if err != nil {
		return nil, fmt.Errorf("GET /api/event: %w", err)
	}
	if resp.StatusCode != http.StatusOK {
		resp.Body.Close()
		return nil, fmt.Errorf("GET /api/event: HTTP %d", resp.StatusCode)
	}
	ch := make(chan Event, 64)
	go func() {
		defer close(ch)
		defer resp.Body.Close()
		readEvents(resp.Body, func(e Event) bool {
			select {
			case ch <- e:
				return true
			case <-ctx.Done():
				return false
			}
		})
	}()
	return ch, nil
}

func readEvents(r io.Reader, emit func(Event) bool) {
	sc := bufio.NewScanner(r)
	sc.Buffer(make([]byte, 64*1024), 16*1024*1024)
	var data strings.Builder
	flush := func() bool {
		defer data.Reset()
		if data.Len() == 0 {
			return true
		}
		if e, ok := parseEvent(data.String()); ok {
			return emit(e)
		}
		return true
	}
	for sc.Scan() {
		line := sc.Text()
		switch {
		case line == "":
			if !flush() {
				return
			}
		case strings.HasPrefix(line, "data:"):
			if data.Len() > 0 {
				data.WriteByte('\n')
			}
			data.WriteString(strings.TrimPrefix(strings.TrimPrefix(line, "data:"), " "))
		}
	}
	flush()
}

func parseEvent(s string) (Event, bool) {
	var raw struct {
		Type string `json:"type"`
		Data struct {
			SessionID string `json:"sessionID"`
			Form      struct {
				SessionID string `json:"sessionID"`
			} `json:"form"`
		} `json:"data"`
	}
	if err := json.Unmarshal([]byte(s), &raw); err != nil || raw.Type == "" {
		return Event{}, false
	}
	id := raw.Data.SessionID
	if id == "" {
		id = raw.Data.Form.SessionID
	}
	return Event{Type: raw.Type, SessionID: id}, true
}

// Relevant reports whether an event can change a task's live state.
func Relevant(t string) bool {
	switch t {
	case "session.status", "session.idle", "session.deleted":
		return true
	}
	return strings.HasPrefix(t, "session.execution.") || strings.HasPrefix(t, "form.") || strings.HasPrefix(t, "permission.")
}

// Settled reports whether an event means the session stopped running, after
// which the agent may have updated its task in Beads.
func Settled(t string) bool {
	switch t {
	case "session.idle", "session.execution.succeeded", "session.execution.failed", "session.execution.interrupted":
		return true
	}
	return false
}

type Herdr interface {

	CreateWorkspace(ctx context.Context, cwd, label string) (string, error)

type CLI struct{ Exec execx.Runner }

// New returns a CLI that runs the herdr binary on PATH.
func New() CLI { return CLI{Exec: execx.Command("herdr")} }

type pane struct {
	PaneID       string `json:"pane_id"`
	WorkspaceID  string `json:"workspace_id"`
	Cwd          string `json:"cwd"`
	AgentSession *struct {
		Value string `json:"value"`
	} `json:"agent_session"`

func (h CLI) call(ctx context.Context, out any, args ...string) error {
	data, err := h.Exec(ctx, "", args...)
	if err != nil {
		return err
	}
	var env struct {
		Result json.RawMessage `json:"result"`
		Error  json.RawMessage `json:"error"`
	}
	if err := json.Unmarshal(data, &env); err != nil {
		return fmt.Errorf("herdr %s: unexpected output: %w", strings.Join(args, " "), err)
	}
	if len(env.Error) > 0 && string(env.Error) != "null" {
		return fmt.Errorf("herdr %s: %s", strings.Join(args, " "), env.Error)
	}
	return json.Unmarshal(env.Result, out)
}

func within(path, root string) bool {
	path = filepath.Clean(path)
	return path == root || strings.HasPrefix(path, root+string(filepath.Separator))
}

// WorkspaceFor returns a workspace with a pane in repo, preferring a pane at
// the repository root over one in a subdirectory or worktree.
func (h CLI) WorkspaceFor(ctx context.Context, repo string) (string, bool, error) {
	var res struct {
		Panes []pane `json:"panes"`
	}
	if err := h.call(ctx, &res, "pane", "list"); err != nil {
		return "", false, err
	}
	repo = filepath.Clean(repo)
	var exact, inside []string
	for _, p := range res.Panes {
		switch {
		case p.Cwd == "":
		case filepath.Clean(p.Cwd) == repo:
			exact = append(exact, p.WorkspaceID)
		case within(p.Cwd, repo):
			inside = append(inside, p.WorkspaceID)
		}
	}
	for _, ids := range [][]string{exact, inside} {
		if len(ids) > 0 {
			sort.Strings(ids)
			return ids[0], true, nil
		}
	}
	return "", false, nil
}

func (h CLI) CreateWorkspace(ctx context.Context, cwd, label string) (string, error) {
	var res struct {
		Workspace struct {
			ID string `json:"workspace_id"`
		} `json:"workspace"`
	}
	if err := h.call(ctx, &res, "workspace", "create", "--cwd", cwd, "--label", label, "--no-focus"); err != nil {
		return "", err
	}
	return res.Workspace.ID, nil
}

func (h CLI) CreateTab(ctx context.Context, workspaceID, cwd, label string) (string, error) {
	var res struct {
		RootPane pane `json:"root_pane"`
	}
	if err := h.call(ctx, &res, "tab", "create", "--workspace", workspaceID, "--cwd", cwd, "--label", label, "--no-focus"); err != nil {
		return "", err
	}
	return res.RootPane.PaneID, nil
}

func (h CLI) Run(ctx context.Context, paneID string, command ...string) error {
	_, err := h.Exec(ctx, "", append([]string{"pane", "run", paneID}, command...)...)
	return err
}

func (h CLI) PaneForSession(ctx context.Context, sessionID string) (string, bool, error) {
	var res struct {
		Agents []pane `json:"agents"`
	}
	if err := h.call(ctx, &res, "agent", "list"); err != nil {
		return "", false, err
	}
	for _, a := range res.Agents {
		if a.AgentSession != nil && a.AgentSession.Value == sessionID {
			return a.PaneID, true, nil
		}
	}
	return "", false, nil
}

func (h CLI) Focus(ctx context.Context, paneID string) error {
	_, err := h.Exec(ctx, "", "agent", "focus", paneID)
	return err
}

// Package gitwt is the only caller of git: per-task worktrees and branches.
package gitwt

import (
	"context"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"strings"

	"github.com/AntoineGS/taskestrator/internal/execx"
)

// Added reports what AddWorktree created, so an undo removes only that.
type Added struct {
	Worktree bool
	Branch   bool
}

// Git manages task worktrees.
type Git interface {
	DefaultBranch(ctx context.Context, repo string) (string, error)
	AddWorktree(ctx context.Context, repo, path, branch, base string) (Added, error)
	RemoveWorktree(ctx context.Context, repo, path, branch string, deleteBranch bool) error
	IsIgnored(ctx context.Context, repo, rel string) (bool, error)
}

// CLI implements Git with the git binary.
type CLI struct{ Exec execx.Runner }

// New returns a CLI that runs git from PATH.
func New() CLI { return CLI{Exec: execx.Command("git")} }

// DefaultBranch returns the branch origin/HEAD points to, or "main".
func (g CLI) DefaultBranch(ctx context.Context, repo string) (string, error) {
	out, err := g.Exec(ctx, repo, "symbolic-ref", "--quiet", "--short", "refs/remotes/origin/HEAD")
	if err != nil {
		if execx.ExitCode(err) > 0 {
			return "main", nil
		}
		return "", err
	}
	ref := strings.TrimSpace(string(out))
	if _, branch, ok := strings.Cut(ref, "/"); ok {
		return branch, nil
	}
	return ref, nil
}

// AddWorktree creates path on branch, creating branch from base when needed.
// An existing worktree already on branch is reused.
func (g CLI) AddWorktree(ctx context.Context, repo, path, branch, base string) (Added, error) {
	if _, err := os.Stat(path); err == nil {
		out, err := g.Exec(ctx, path, "rev-parse", "--abbrev-ref", "HEAD")
		if err != nil {
			return Added{}, fmt.Errorf("%s exists but is not a git worktree: %w", path, err)
		}
		if cur := strings.TrimSpace(string(out)); cur != branch {
			return Added{}, fmt.Errorf("%s exists on branch %s, expected %s", path, cur, branch)
		}
		return Added{}, nil
	} else if !errors.Is(err, fs.ErrNotExist) {
		return Added{}, err
	}
	_, err := g.Exec(ctx, repo, "rev-parse", "--verify", "--quiet", "refs/heads/"+branch)
	switch {
	case err == nil:
		if _, err := g.Exec(ctx, repo, "worktree", "add", path, branch); err != nil {
			return Added{}, err
		}
		return Added{Worktree: true}, nil
	case execx.ExitCode(err) == 1:
		if _, err := g.Exec(ctx, repo, "worktree", "add", "-b", branch, path, base); err != nil {
			return Added{}, err
		}
		return Added{Worktree: true, Branch: true}, nil
	default:
		return Added{}, err
	}
}

// RemoveWorktree removes path and, when deleteBranch is set, its branch.
func (g CLI) RemoveWorktree(ctx context.Context, repo, path, branch string, deleteBranch bool) error {
	if _, err := g.Exec(ctx, repo, "worktree", "remove", "--force", path); err != nil {
		return err
	}
	if deleteBranch {
		if _, err := g.Exec(ctx, repo, "branch", "-D", branch); err != nil {
			return err
		}
	}
	return nil
}

// IsIgnored reports whether rel is git-ignored in repo.
func (g CLI) IsIgnored(ctx context.Context, repo, rel string) (bool, error) {
	_, err := g.Exec(ctx, repo, "check-ignore", "--quiet", rel)
	if err == nil {
		return true, nil
	}
	if execx.ExitCode(err) == 1 {
		return false, nil
	}
	return false, err
}

package dispatch

import (
	"fmt"
	"strings"
	"text/template"
)

// DefaultPrompt is used when a project's .tsk.toml sets no prompt.
const DefaultPrompt = `You are working on task {{.ID}}: {{.Title}}
{{if .Description}}
## Description

{{.Description}}
{{end}}{{if .Acceptance}}
## Acceptance criteria

{{.Acceptance}}
{{end}}
## Rules

- Work only in this worktree: {{.Worktree}} (branch {{.Branch}}).
- Commit on this branch with a "Task: {{.ID}}" trailer. Never push, merge or rebase onto another branch.
- If you need input from me, ask with the question tool.
- When the work is done and verified, run:
  bd update {{.ID}} --status review --notes "<one-paragraph summary of what changed and how it was verified>"
`

// PromptData is the data available to prompt templates.
type PromptData struct {
	ID, Title, Description, Acceptance, Worktree, Branch string
}

// RenderPrompt renders tmpl, or DefaultPrompt when tmpl is blank.
func RenderPrompt(tmpl string, data PromptData) (string, error) {
	if strings.TrimSpace(tmpl) == "" {
		tmpl = DefaultPrompt
	}
	t, err := template.New("prompt").Parse(tmpl)
	if err != nil {
		return "", fmt.Errorf("prompt template: %w", err)
	}
	var b strings.Builder
	if err := t.Execute(&b, data); err != nil {
		return "", fmt.Errorf("prompt template: %w", err)
	}
	return b.String(), nil
}

// Package dispatch starts an OpenCode agent on a Beads task.
package dispatch

import (
	"context"
	"errors"
	"fmt"
	"path/filepath"

	"github.com/AntoineGS/taskestrator/internal/config"
	"github.com/AntoineGS/taskestrator/internal/discover"
	"github.com/AntoineGS/taskestrator/internal/execx"

	"github.com/AntoineGS/taskestrator/internal/opencode"
	"github.com/AntoineGS/taskestrator/internal/tracker"
)

// OpenCode is the part of the OpenCode client dispatch uses.
type OpenCode interface {
	CreateSession(ctx context.Context, in opencode.CreateSession) (opencode.Session, error)
	DeleteSession(ctx context.Context, id string) error
	Prompt(ctx context.Context, id, text string) error
}

// Deps are dispatch's collaborators. Shell runs "sh".
type Deps struct {
	Tracker  tracker.Tracker
	OpenCode OpenCode
	Herdr    herdr.Herdr
	Git      gitwt.Git
	Shell    execx.Runner
}

// Request is one task to dispatch.
type Request struct {
	Project discover.Project
	Task    tracker.Task
	File    config.ProjectFile
}

// Result describes a dispatched task. TabError is set when the agent started
// but its herdr tab could not be opened.
type Result struct {
	SessionID string
	Worktree  string
	Branch    string
	PaneID    string
	TabError  error
}

// Run performs the spec's seven dispatch steps. On a failure before the
// prompt is sent, completed steps are undone in reverse order.
func Run(ctx context.Context, d Deps, req Request) (Result, error) {
	repo, id := req.Project.Repo, req.Task.ID
	branch := "task/" + id
	wt := filepath.Join(repo, ".worktrees", id)
	text, err := RenderPrompt(req.File.Prompt, PromptData{
		ID: id, Title: req.Task.Title, Description: req.Task.Description,
		Acceptance: req.Task.Acceptance, Worktree: wt, Branch: branch,
	})
	if err != nil {
		return Result{}, err
	}

	var undo []func(context.Context) error
	fail := func(step string, err error) (Result, error) {
		errs := []error{fmt.Errorf("%s: %w", step, err)}
		uctx := context.WithoutCancel(ctx)
		for i := len(undo) - 1; i >= 0; i-- {
			if uerr := undo[i](uctx); uerr != nil {
				errs = append(errs, fmt.Errorf("undo: %w", uerr))
			}
		}
		return Result{}, errors.Join(errs...)
	}

	if err := d.Tracker.Claim(ctx, repo, id); err != nil {
		return fail("claim", err)
	}
	undo = append(undo, func(c context.Context) error { return d.Tracker.Unclaim(c, repo, id) })

	base, err := d.Git.DefaultBranch(ctx, repo)
	if err != nil {
		return fail("default branch", err)
	}
	added, err := d.Git.AddWorktree(ctx, repo, wt, branch, base)
	if err != nil {
		return fail("worktree", err)
	}
	if added.Worktree {
		undo = append(undo, func(c context.Context) error {
			return d.Git.RemoveWorktree(c, repo, wt, branch, added.Branch)
		})
	}

	if req.File.Setup != "" {
		if _, err := d.Shell(ctx, wt, "-c", req.File.Setup); err != nil {
			return fail("setup", err)
		}
	}

	sess, err := d.OpenCode.CreateSession(ctx, opencode.CreateSession{
		Title: id + ": " + req.Task.Title, Directory: wt, Metadata: map[string]string{"tsk.task": id},
	})
	if err != nil {
		return fail("session", err)
	}
	undo = append(undo, func(c context.Context) error { return d.OpenCode.DeleteSession(c, sess.ID) })

	meta := map[string]string{tracker.MetaSession: sess.ID, tracker.MetaWorktree: wt, tracker.MetaBranch: branch}
	if err := d.Tracker.SetMetadata(ctx, repo, id, meta); err != nil {
		return fail("record", err)
	}
	undo = append(undo, func(c context.Context) error {
		return d.Tracker.UnsetMetadata(c, repo, id, tracker.MetaSession, tracker.MetaWorktree, tracker.MetaBranch)
	})

	if err := d.OpenCode.Prompt(ctx, sess.ID, text); err != nil {
		return fail("prompt", err)
	}

	res := Result{SessionID: sess.ID, Worktree: wt, Branch: branch}
	pane, err := OpenTab(ctx, d.Herdr, Target{
		Repo: repo, Project: req.Project.Name, Worktree: wt, TaskID: id, Title: req.Task.Title, SessionID: sess.ID,
	})
	if err != nil {
		res.TabError = err
		return res, nil
	}
	res.PaneID = pane
	return res, nil
}

package dispatch

import (
	"context"
	"errors"
	"fmt"
	"os"

	"github.com/AntoineGS/taskestrator/internal/discover"
	"github.com/AntoineGS/taskestrator/internal/herdr"
	"github.com/AntoineGS/taskestrator/internal/tracker"
)

// Target identifies a dispatched task's session and where it runs.
type Target struct {
	Repo, Project, Worktree, TaskID, Title, SessionID string
}

// TargetFor builds a Target from a dispatched task.
func TargetFor(p discover.Project, t tracker.Task) Target {
	return Target{
		Repo: p.Repo, Project: p.Name, Worktree: t.Meta(tracker.MetaWorktree),
		TaskID: t.ID, Title: t.Title, SessionID: t.Meta(tracker.MetaSession),
	}
}

// TabLabel is the herdr tab label for a task: its ID and up to 30 characters of title.
func TabLabel(id, title string) string {
	const max = 30
	if r := []rune(title); len(r) > max {
		title = string(r[:max-1]) + "…"
	}
	return id + " " + title
}

// OpenTab opens a tab in the repository's workspace, creating the workspace
// if needed, and attaches OpenCode to the task's session.
func OpenTab(ctx context.Context, h herdr.Herdr, t Target) (string, error) {
	ws, ok, err := h.WorkspaceFor(ctx, t.Repo)
	if err != nil {
		return "", err
	}
	if !ok {
		if ws, err = h.CreateWorkspace(ctx, t.Repo, t.Project); err != nil {
			return "", err
		}
	}
	pane, err := h.CreateTab(ctx, ws, t.Worktree, TabLabel(t.TaskID, t.Title))
	if err != nil {
		return "", err
	}
	if err := h.Run(ctx, pane, "opencode", "--session", t.SessionID); err != nil {
		return "", err
	}
	return pane, nil
}

// Jump focuses the pane running the task's session, opening one when none exists.
func Jump(ctx context.Context, h herdr.Herdr, t Target) error {
	if t.SessionID == "" {
		return errors.New("task has not been dispatched")
	}
	pane, ok, err := h.PaneForSession(ctx, t.SessionID)
	if err != nil {
		return err
	}
	if !ok {
		if _, err := os.Stat(t.Worktree); err != nil {
			return fmt.Errorf("worktree %s is missing: %w", t.Worktree, err)
		}
		if pane, err = OpenTab(ctx, h, t); err != nil {
			return err
		}
	}
	return h.Focus(ctx, pane)
}

// Package board holds the board's in-memory view of tasks and their live state.
package board

import (
	"sort"
	"time"

	"github.com/AntoineGS/taskestrator/internal/discover"
	"github.com/AntoineGS/taskestrator/internal/status"
	"github.com/AntoineGS/taskestrator/internal/tracker"
)

// Row is one task as the board shows it.
type Row struct {
	Project discover.Project
	Task    tracker.Task
	Display status.Display
	Since   time.Time
}

type key struct{ repo, id string }

type entry struct {
	project discover.Project
	task    tracker.Task
	display status.Display
	since   time.Time
}

// Store combines tasks from every project with session snapshots. It is not
// safe for concurrent use; the Bubble Tea model owns it.
type Store struct {
	now       func() time.Time
	reachable bool
	entries   map[key]*entry
	snaps     map[string]*status.Snapshot
}

// NewStore returns an empty store that uses now for timestamps.
func NewStore(now func() time.Time) *Store {
	return &Store{now: now, entries: map[key]*entry{}, snaps: map[string]*status.Snapshot{}}
}

func (s *Store) derive(t tracker.Task) status.Display {
	sid := t.Meta(tracker.MetaSession)
	var snap *status.Snapshot
	if s.reachable {
		snap = s.snaps[sid]
	}
	return status.Derive(t.Status, sid, snap)
}

// SetTasks replaces one project's tasks.
func (s *Store) SetTasks(p discover.Project, tasks []tracker.Task) {
	old := map[string]*entry{}
	for k, e := range s.entries {
		if k.repo == p.Repo {
			old[k.id] = e
			delete(s.entries, k)
		}
	}
	for _, t := range tasks {
		e := &entry{project: p, task: t, display: s.derive(t)}
		switch o, ok := old[t.ID]; {
		case ok && o.display == e.display:
			e.since = o.since
		case ok:
			e.since = s.now()
		default:
			e.since = t.UpdatedAt
		}
		s.entries[key{p.Repo, t.ID}] = e
	}
}

func (s *Store) recompute() {
	for _, e := range s.entries {
		if d := s.derive(e.task); d != e.display {
			e.display = d
			e.since = s.now()
		}
	}
}

// SetSnapshot records a session's live state; nil means it could not be loaded.
func (s *Store) SetSnapshot(sessionID string, snap *status.Snapshot) {
	s.snaps[sessionID] = snap
	s.recompute()
}

// SetReachable records whether the OpenCode service answered.
func (s *Store) SetReachable(ok bool) {
	s.reachable = ok
	s.recompute()
}

// Reachable reports whether the OpenCode service last answered.
func (s *Store) Reachable() bool { return s.reachable }

var order = map[status.State]int{
	status.Waiting: 0, status.InProgress: 1, status.Review: 2, status.Lost: 3,
	status.Unknown: 4, status.Queued: 5, status.Done: 6,
}

// Rows returns dispatched, non-closed tasks grouped by state. With showAll it
// adds queued tasks and tasks closed in the last 24 hours.
func (s *Store) Rows(showAll bool) []Row {
	cutoff := s.now().Add(-24 * time.Hour)
	var rows []Row
	for _, e := range s.entries {
		switch e.display.State {
		case status.Untracked:
			continue
		case status.Done:
			if !showAll || e.task.ClosedAt == nil || e.task.ClosedAt.Before(cutoff) {
				continue
			}
		case status.Queued:
			if !showAll {
				continue
			}
		default:
			if e.task.Meta(tracker.MetaSession) == "" {
				continue
			}
		}
		rows = append(rows, Row{Project: e.project, Task: e.task, Display: e.display, Since: e.since})
	}
	sort.Slice(rows, func(i, j int) bool {
		a, b := rows[i], rows[j]
		if order[a.Display.State] != order[b.Display.State] {
			return order[a.Display.State] < order[b.Display.State]
		}
		if !a.Since.Equal(b.Since) {
			return a.Since.Before(b.Since)
		}
		if a.Project.Name != b.Project.Name {
			return a.Project.Name < b.Project.Name
		}
		return a.Task.ID < b.Task.ID
	})
	return rows
}

// Queued counts queued tasks per project name.
func (s *Store) Queued() map[string]int {
	counts := map[string]int{}
	for _, e := range s.entries {
		if e.display.State == status.Queued {
			counts[e.project.Name]++
		}
	}
	return counts
}

func live(t tracker.Task) bool {
	return t.Meta(tracker.MetaSession) != "" && t.Status != "closed" && t.Status != tracker.StatusReview
}

// Sessions returns the session IDs of dispatched tasks that are neither closed nor in review.
func (s *Store) Sessions() []string { return s.sessions("") }

// SessionsFor is Sessions limited to one repository.
func (s *Store) SessionsFor(repo string) []string { return s.sessions(repo) }

func (s *Store) sessions(repo string) []string {
	var out []string
	for k, e := range s.entries {
		if (repo == "" || k.repo == repo) && live(e.task) {
			out = append(out, e.task.Meta(tracker.MetaSession))
		}
	}
	sort.Strings(out)
	return out
}

// ProjectForSession returns the project of the task dispatched with sessionID.
func (s *Store) ProjectForSession(sessionID string) (discover.Project, bool) {
	for _, e := range s.entries {
		if e.task.Meta(tracker.MetaSession) == sessionID {
			return e.project, true
		}
	}
	return discover.Project{}, false
}

// collect runs cmd and returns the messages it produces, expanding batches.
// Commands that block (event waits) are dropped after a short timeout.
func collect(cmd tea.Cmd) []tea.Msg {
	if cmd == nil {
		return nil
	}
	done := make(chan tea.Msg, 1)
	go func() { done <- cmd() }()
	select {
	case msg := <-done:
		if b, ok := msg.(tea.BatchMsg); ok {
			var out []tea.Msg
			for _, c := range b {
				out = append(out, collect(c)...)
			}
			return out
		}
		if msg == nil {
			return nil
		}
		return []tea.Msg{msg}
	case <-time.After(50 * time.Millisecond):
		return nil
	}
}

// settle feeds every resulting message back into the model until none remain.
func settle(t *testing.T, m tea.Model, cmd tea.Cmd) (tea.Model, []tea.Msg) {
	t.Helper()
	var seen []tea.Msg
	queue := collect(cmd)
	for i := 0; len(queue) > 0; i++ {
		if i > 500 {
			t.Fatal("model did not settle")
		}
		msg := queue[0]
		queue = queue[1:]
		seen = append(seen, msg)
		var c tea.Cmd
		m, c = m.Update(msg)
		queue = append(queue, collect(c)...)
	}
	return m, seen
}

// loadSnapshots fetches the live state of each session. A session whose
// snapshot fails is recorded as nil, which shows as unknown.
func loadSnapshots(ctx context.Context, live Live, ids []string) snapsMsg {
	if live == nil {
		return snapsMsg{reachable: false}
	}
	ctx, cancel := context.WithTimeout(ctx, requestTimeout)
	defer cancel()
	active, err := live.Active(ctx)
	if err != nil {
		return snapsMsg{reachable: false, err: err}
	}
	snaps := make(map[string]*status.Snapshot, len(ids))
	for _, id := range ids {
		snap, err := live.Snapshot(ctx, id, active)
		if err != nil {
			snaps[id] = nil
			continue
		}
		snaps[id] = &snap
	}
	return snapsMsg{snaps: snaps, reachable: true}
}

	"context"
	"fmt"
	"sort"
	"strings"
	"time"

	"github.com/AntoineGS/taskestrator/internal/board"
	"github.com/AntoineGS/taskestrator/internal/discover"

// BoardConfig configures NewBoard. Live and Jump are nil when OpenCode or
// herdr is unavailable.
type BoardConfig struct {
	Projects []discover.Project
	Tracker  tracker.Tracker
	Live     Live
	Jump     Jumper
	Reread   time.Duration
	Now      func() time.Time
	After    After
	Warnings []string
}

// Board is the global task board.
type Board struct {
	ctx      context.Context
	projects []discover.Project
	tracker  tracker.Tracker
	live     Live
	jump     Jumper
	reread   time.Duration
	now      func() time.Time
	after    After
	store    *board.Store

	cursor   int
	showAll  bool
	filter   string
	editing  bool
	warnings []string
	errs     map[string]string
	liveErr  string
	message  string
	pending  map[string]bool
	backoff  time.Duration
}

type tasksMsg struct {
	project discover.Project
	tasks   []tracker.Task
	err     error
}

type subscribedMsg struct {
	ch  <-chan opencode.Event
	err error
}

type eventMsg struct {
	ev opencode.Event
	ch <-chan opencode.Event
	ok bool
}

type rootMsg struct {
	root    string
	settled bool
}

type refreshMsg struct{ id string }

type reconnectMsg struct{}

// NewBoard returns the board model.
func NewBoard(ctx context.Context, cfg BoardConfig) *Board {
	if cfg.Now == nil {
		cfg.Now = time.Now
	}
	if cfg.After == nil {
		cfg.After = realAfter
	}
	if cfg.Reread <= 0 {
		cfg.Reread = 10 * time.Second
	}
	return &Board{
		ctx: ctx, projects: cfg.Projects, tracker: cfg.Tracker, live: cfg.Live, jump: cfg.Jump,
		reread: cfg.Reread, now: cfg.Now, after: cfg.After, store: board.NewStore(cfg.Now),
		warnings: cfg.Warnings, errs: map[string]string{}, pending: map[string]bool{},
	}
}

func (b *Board) Init() tea.Cmd {
	cmds := []tea.Cmd{b.after(b.reread, tickMsg{})}
	for _, p := range b.projects {
		cmds = append(cmds, b.loadTasks(p))
	}
	if b.live != nil {
		cmds = append(cmds, b.subscribe())
	} else {
		b.liveErr = "OpenCode service unavailable"
	}
	return tea.Batch(cmds...)
}

func (b *Board) loadTasks(p discover.Project) tea.Cmd {
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(b.ctx, requestTimeout)
		defer cancel()
		tasks, err := b.tracker.List(ctx, p.Repo)
		return tasksMsg{project: p, tasks: tasks, err: err}
	}
}

func (b *Board) loadSnaps(ids []string) tea.Cmd {
	return func() tea.Msg { return loadSnapshots(b.ctx, b.live, ids) }
}

func (b *Board) subscribe() tea.Cmd {
	return func() tea.Msg {
		ch, err := b.live.Subscribe(b.ctx)
		return subscribedMsg{ch: ch, err: err}
	}
}

func waitEvent(ch <-chan opencode.Event) tea.Cmd {
	return func() tea.Msg {
		ev, ok := <-ch
		return eventMsg{ev: ev, ch: ch, ok: ok}
	}
}

func (b *Board) resolveRoot(ev opencode.Event) tea.Cmd {
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(b.ctx, requestTimeout)
		defer cancel()
		root, err := b.live.Root(ctx, ev.SessionID)
		if err != nil {
			return nil
		}
		return rootMsg{root: root, settled: opencode.Settled(ev.Type)}
	}

func (b *Board) scheduleReconnect() tea.Cmd {
	if b.backoff == 0 {
		b.backoff = time.Second
	} else if b.backoff < 30*time.Second {
		b.backoff *= 2
	}
	return b.after(b.backoff, reconnectMsg{})
}

func (b *Board) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		return b.key(msg)
	case tasksMsg:
		if msg.err != nil {
			b.errs[msg.project.Name] = msg.err.Error()
			return b, nil
		}
		delete(b.errs, msg.project.Name)
		b.store.SetTasks(msg.project, msg.tasks)
		b.refreshRows()
		return b, b.loadSnaps(b.store.SessionsFor(msg.project.Repo))
	case snapsMsg:
		if msg.reachable {
			b.liveErr = ""
			for id, s := range msg.snaps {
				b.store.SetSnapshot(id, s)
			}
			b.store.SetReachable(true)
		} else {
			if msg.err != nil {
				b.liveErr = "OpenCode service unreachable: " + msg.err.Error()
			}
			b.store.SetReachable(false)
		}
		b.refreshRows()
		return b, nil
	case subscribedMsg:
		if msg.err != nil {
			b.liveErr = "OpenCode event stream: " + msg.err.Error()
			return b, b.scheduleReconnect()
		}
		b.backoff = 0
		return b, tea.Batch(waitEvent(msg.ch), b.loadSnaps(b.store.Sessions()))
	case eventMsg:
		if !msg.ok {
			b.liveErr = "OpenCode event stream disconnected; reconnecting"
			return b, b.scheduleReconnect()
		}
		var cmds []tea.Cmd
		if msg.ch != nil {
			cmds = append(cmds, waitEvent(msg.ch))
		}
		if msg.ev.SessionID != "" && opencode.Relevant(msg.ev.Type) {
			cmds = append(cmds, b.resolveRoot(msg.ev))
		}
		return b, tea.Batch(cmds...)
	case rootMsg:
		p, ok := b.store.ProjectForSession(msg.root)
		if !ok {
			return b, nil
		}
		var cmds []tea.Cmd
		if msg.settled {
			cmds = append(cmds, b.loadTasks(p))
		}
		if !b.pending[msg.root] {
			b.pending[msg.root] = true
			refresh := refreshMsg{id: msg.root}
			if c := b.after(300*time.Millisecond, refresh); c != nil {
				cmds = append(cmds, c)
			} else {
				cmds = append(cmds, func() tea.Msg { return refresh })
			}
		}
		return b, tea.Batch(cmds...)
	case refreshMsg:
		delete(b.pending, msg.id)
		return b, b.loadSnaps([]string{msg.id})
	case tickMsg:
		cmds := []tea.Cmd{b.after(b.reread, tickMsg{})}
		for _, p := range b.projects {
			cmds = append(cmds, b.loadTasks(p))
		}
		return b, tea.Batch(cmds...)
	case reconnectMsg:
		return b, b.subscribe()
	case jumpedMsg:
		if msg.err != nil {
			b.message = "jump failed: " + msg.err.Error()
			return b, nil
		}
		return b, tea.Quit
	}
	return b, nil
}

func (b *Board) key(msg tea.KeyPressMsg) (tea.Model, tea.Cmd) {
	k := msg.String()
	if b.editing {
		switch k {
		case "enter":
			b.editing = false
		case "esc":
			b.editing, b.filter = false, ""
		case "backspace":
			if r := []rune(b.filter); len(r) > 0 {
				b.filter = string(r[:len(r)-1])
			}
		default:
			b.filter += msg.Text
		}
		b.refreshRows()
		return b, nil
	}
	switch k {
	case "q", "esc", "ctrl+c":
		return b, tea.Quit
	case "j", "down":
		if b.cursor < len(b.rows)-1 {
			b.cursor++
		}
	case "k", "up":
		if b.cursor > 0 {
			b.cursor--
		}
	case "a":
		b.showAll = !b.showAll
		b.refreshRows()
	case "/":
		b.editing = true
	case "enter":
		return b, b.enter()
	}
	return b, nil
}

func (b *Board) enter() tea.Cmd {
	if len(b.rows) == 0 {
		return nil
	}
	row := b.rows[b.cursor]
	switch {
	case row.Display.State == status.Lost:
		b.message = row.Task.ID + ": its session is gone; dispatch it again with d in `tsk project`"
		return nil
	case row.Task.Meta(tracker.MetaSession) == "":
		b.message = row.Task.ID + " has not been dispatched; use `tsk project`"
		return nil
	case b.jump == nil:
		b.message = "herdr is not available, so tsk cannot jump to tasks"
		return nil
	}
	target := dispatch.TargetFor(row.Project, row.Task)
	jump := b.jump
	ctx := b.ctx
	return func() tea.Msg { return jumpedMsg{err: jump(ctx, target)} }
}

func (b *Board) refreshRows() {
	rows := b.store.Rows(b.showAll)
	if f := strings.ToLower(b.filter); f != "" {
		kept := rows[:0]
		for _, r := range rows {
			hay := strings.ToLower(r.Task.ID + " " + r.Task.Title + " " + r.Project.Name)
			if strings.Contains(hay, f) {
				kept = append(kept, r)
			}
		}
		rows = kept
	}
	b.rows = rows
	if b.cursor >= len(rows) {
		b.cursor = max(0, len(rows)-1)
	}
}

func (b *Board) View() tea.View {
	var s strings.Builder
	counts := map[status.State]int{}
	for _, r := range b.store.Rows(false) {
		counts[r.Display.State]++
	}
	fmt.Fprintf(&s, "tsk · %d waiting · %d in progress · %d review", counts[status.Waiting], counts[status.InProgress], counts[status.Review])
	if q := b.store.Queued(); len(q) > 0 {
		names := make([]string, 0, len(q))
		for n := range q {
			names = append(names, n)
		}
		sort.Strings(names)
		parts := make([]string, 0, len(names))
		for _, n := range names {
			parts = append(parts, fmt.Sprintf("%s %d", n, q[n]))
		}
		fmt.Fprintf(&s, "    queued: %s", strings.Join(parts, ", "))
	}
	s.WriteString("\n")
	if b.liveErr != "" {
		fmt.Fprintf(&s, "! %s\n", b.liveErr)
	}
	for _, w := range b.warnings {
		fmt.Fprintf(&s, "! %s\n", w)
	}
	names := make([]string, 0, len(b.errs))
	for n := range b.errs {
		names = append(names, n)
	}
	sort.Strings(names)
	for _, n := range names {
		fmt.Fprintf(&s, "! %s: %s\n", n, b.errs[n])
	}
	if b.editing || b.filter != "" {
		fmt.Fprintf(&s, "filter: %s\n", b.filter)
	}
	s.WriteString("\n")
	if len(b.rows) == 0 {
		s.WriteString("  nothing to show\n")
	}
	group := status.State(-1)
	for i, r := range b.rows {
		if r.Display.State != group {
			group = r.Display.State
			fmt.Fprintf(&s, "%s\n", strings.ToUpper(group.String()))
		}
		cursor := "  "
		if i == b.cursor {
			cursor = "> "
		}
		fmt.Fprintf(&s, "%s%-10s %-4s %-40s %-11s %s\n", cursor, r.Task.ID, r.Project.Name,
			truncate(r.Task.Title, 40), r.Display.Detail, age(b.now().Sub(r.Since)))
	}
	if b.message != "" {
		fmt.Fprintf(&s, "\n%s\n", b.message)
	}
	s.WriteString("\nj/k move · enter jump · a all · / filter · q close\n")
	v := tea.NewView(s.String())
	v.AltScreen = true
	return v
}

	tasks     []tracker.Task
	ready     map[string]int
	snaps     map[string]*status.Snapshot
	reachable bool
	rows      []projectRow
	cursor    int

	checked   bool
	reviewOK  bool
	ignoredOK bool
	busy      bool
	loadErr   string
	message   string
}

type projectLoadMsg struct {
	tasks []tracker.Task
	ready []tracker.Task
	err   error
}

type checksMsg struct {
	reviewOK, ignoredOK bool
	err                 error
}

type dispatchedMsg struct {
	id  string
	res dispatch.Result
	err error
}

type reviewAddedMsg struct{ err error }

// NewProject returns the project model.
func NewProject(ctx context.Context, cfg ProjectConfig) *Project {
	if cfg.After == nil {
		cfg.After = realAfter
	}
	if cfg.Reread <= 0 {
		cfg.Reread = 10 * time.Second
	}
	return &Project{ctx: ctx, cfg: cfg, ready: map[string]int{}, snaps: map[string]*status.Snapshot{}}
}

func (p *Project) Init() tea.Cmd {
	return tea.Batch(p.load(), p.checks(), p.cfg.After(p.cfg.Reread, tickMsg{}))
}

func (p *Project) load() tea.Cmd {
	ctx, tr, repo := p.ctx, p.cfg.Tracker, p.cfg.Project.Repo
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(ctx, requestTimeout)
		defer cancel()
		tasks, err := tr.List(ctx, repo)
		if err != nil {
			return projectLoadMsg{err: err}
		}
		ready, err := tr.Ready(ctx, repo)
		return projectLoadMsg{tasks: tasks, ready: ready, err: err}
	}
}

func (p *Project) checks() tea.Cmd {
	ctx, tr, git, repo := p.ctx, p.cfg.Tracker, p.cfg.Git, p.cfg.Project.Repo
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(ctx, requestTimeout)
		defer cancel()
		reviewOK, err := tr.HasReviewStatus(ctx, repo)
		if err != nil {
			return checksMsg{err: err}
		}
		ignoredOK, err := git.IsIgnored(ctx, repo, ".worktrees/tsk-check")
		return checksMsg{reviewOK: reviewOK, ignoredOK: ignoredOK, err: err}
	}
}

func (p *Project) sessions() []string {
	var ids []string
	for _, t := range p.tasks {
		if id := t.Meta(tracker.MetaSession); id != "" && t.Status != "closed" && t.Status != tracker.StatusReview {
			ids = append(ids, id)
		}
	}
	return ids
}

func (p *Project) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		return p.key(msg)
	case projectLoadMsg:
		if msg.err != nil {
			p.loadErr = msg.err.Error()
			return p, nil
		}
		p.loadErr = ""
		p.tasks = msg.tasks
		p.ready = map[string]int{}
		for i, t := range msg.ready {
			p.ready[t.ID] = i
		}
		p.rebuild()
		ctx, live, ids := p.ctx, p.cfg.Live, p.sessions()
		return p, func() tea.Msg { return loadSnapshots(ctx, live, ids) }
	case snapsMsg:
		p.reachable = msg.reachable
		if msg.reachable {
			for id, s := range msg.snaps {
				p.snaps[id] = s
			}
		}
		p.rebuild()
		return p, nil
	case checksMsg:
		p.checked = true
		if msg.err != nil {
			p.message = "setup check failed: " + msg.err.Error()
		}
		p.reviewOK, p.ignoredOK = msg.reviewOK, msg.ignoredOK
		return p, nil
	case reviewAddedMsg:
		if msg.err != nil {
			p.message = "could not add the review status: " + msg.err.Error()
			return p, nil
		}
		p.message = "added the review status"
		return p, p.checks()
	case dispatchedMsg:
		p.busy = false
		switch {
		case msg.err != nil:
			p.message = "dispatch failed: " + msg.err.Error()
		case msg.res.TabError != nil:
			p.message = fmt.Sprintf("dispatched %s; the agent is working but tsk could not open its tab: %v", msg.id, msg.res.TabError)
		default:
			p.message = "dispatched " + msg.id
		}
		return p, p.load()
	case tickMsg:
		return p, tea.Batch(p.load(), p.cfg.After(p.cfg.Reread, tickMsg{}))
	case jumpedMsg:
		if msg.err != nil {
			p.message = "jump failed: " + msg.err.Error()
			return p, nil
		}
		return p, tea.Quit
	}
	return p, nil
}

func (p *Project) rebuild() {
	var rows []projectRow
	for _, t := range p.tasks {
		if t.Status == "closed" {
			continue
		}
		sid := t.Meta(tracker.MetaSession)
		var snap *status.Snapshot
		if p.reachable {
			snap = p.snaps[sid]
		}
		_, ready := p.ready[t.ID]
		rows = append(rows, projectRow{task: t, display: status.Derive(t.Status, sid, snap), ready: ready})
	}
	sort.SliceStable(rows, func(i, j int) bool {
		a, b := rows[i], rows[j]
		if a.ready != b.ready {
			return a.ready
		}
		if a.ready {
			return p.ready[a.task.ID] < p.ready[b.task.ID]
		}
		if a.display.State != b.display.State {
			return a.display.State < b.display.State
		}
		return a.task.ID < b.task.ID
	})
	p.rows = rows
	if p.cursor >= len(rows) {
		p.cursor = max(0, len(rows)-1)
	}
}

func (p *Project) key(msg tea.KeyPressMsg) (tea.Model, tea.Cmd) {
	switch msg.String() {
	case "q", "esc", "ctrl+c":
		return p, tea.Quit
	case "j", "down":
		if p.cursor < len(p.rows)-1 {
			p.cursor++
		}
	case "k", "up":
		if p.cursor > 0 {
			p.cursor--
		}
	case "R":
		ctx, tr, repo := p.ctx, p.cfg.Tracker, p.cfg.Project.Repo
		return p, func() tea.Msg { return reviewAddedMsg{err: tr.AddReviewStatus(ctx, repo)} }
	case "d":
		return p, p.dispatch()
	case "enter":
		return p, p.enter()
	}
	return p, nil
}

func (p *Project) dispatch() tea.Cmd {
	if p.busy || len(p.rows) == 0 {
		return nil
	}
	row := p.rows[p.cursor]
	dispatchable := (row.ready && row.display.State == status.Queued) || row.display.State == status.Lost
	switch {
	case !dispatchable:
		p.message = row.task.ID + " is not ready to dispatch (" + row.display.String() + ")"
		return nil
	case !p.reviewOK:
		p.message = "add the review status first: press R"
		return nil
	case !p.ignoredOK:
		p.message = ".worktrees/ is not git-ignored in this repo; add it to .gitignore first"
		return nil
	case p.cfg.Dispatch == nil:
		p.message = "OpenCode or herdr is not available, so tsk cannot dispatch"
		return nil
	}
	p.busy = true
	p.message = "dispatching " + row.task.ID + "…"
	ctx, run := p.ctx, p.cfg.Dispatch
	req := dispatch.Request{Project: p.cfg.Project, Task: row.task, File: p.cfg.File}
	return func() tea.Msg {
		res, err := run(ctx, req)
		return dispatchedMsg{id: req.Task.ID, res: res, err: err}
	}
}

func (p *Project) enter() tea.Cmd {
	if len(p.rows) == 0 {
		return nil
	}
	row := p.rows[p.cursor]
	switch {
	case row.task.Meta(tracker.MetaSession) == "":
		p.message = row.task.ID + " has not been dispatched; press d"
		return nil
	case row.display.State == status.Lost:
		p.message = row.task.ID + ": its session is gone; press d to dispatch it again"
		return nil
	case p.cfg.Jump == nil:
		p.message = "herdr is not available, so tsk cannot jump to tasks"
		return nil
	}
	ctx, jump := p.ctx, p.cfg.Jump
	target := dispatch.TargetFor(p.cfg.Project, row.task)
	return func() tea.Msg { return jumpedMsg{err: jump(ctx, target)} }
}

func (p *Project) View() tea.View {
	var s strings.Builder
	fmt.Fprintf(&s, "%s  (%s)\n", p.cfg.Project.Repo, p.cfg.Project.Name)
	for _, w := range p.cfg.Warnings {
		fmt.Fprintf(&s, "! %s\n", w)
	}
	if p.loadErr != "" {
		fmt.Fprintf(&s, "! %s\n", p.loadErr)
	}
	if p.cfg.Live != nil && !p.reachable {
		s.WriteString("! OpenCode service unreachable; live states are unknown\n")
	}
	if p.checked && !p.reviewOK {
		s.WriteString("! this project has no review status; press R to add it\n")
	}
	if p.checked && !p.ignoredOK {
		s.WriteString("! .worktrees/ is not git-ignored; add it to .gitignore\n")
	}
	s.WriteString("\n")
	if len(p.rows) == 0 {
		s.WriteString("  no open tasks\n")
	}
	header := ""
	for i, r := range p.rows {
		h := "OTHER"
		if r.ready {
			h = "READY"
		}
		if h != header {
			header = h
			fmt.Fprintf(&s, "%s\n", h)
		}
		cursor := "  "
		if i == p.cursor {
			cursor = "> "
		}
		fmt.Fprintf(&s, "%s%-10s P%d  %-40s %s\n", cursor, r.task.ID, r.task.Priority, truncate(r.task.Title, 40), r.display)
	}
	if p.message != "" {
		fmt.Fprintf(&s, "\n%s\n", p.message)
	}
	s.WriteString("\nj/k move · enter jump · d dispatch · R add review status · q quit\n")
	v := tea.NewView(s.String())
	v.AltScreen = true
	return v
}

package main

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"os/signal"
	"path/filepath"

	tea "charm.land/bubbletea/v2"

	"github.com/AntoineGS/taskestrator/internal/config"
	"github.com/AntoineGS/taskestrator/internal/discover"
	"github.com/AntoineGS/taskestrator/internal/dispatch"
	"github.com/AntoineGS/taskestrator/internal/execx"
	"github.com/AntoineGS/taskestrator/internal/gitwt"
	"github.com/AntoineGS/taskestrator/internal/herdr"
	"github.com/AntoineGS/taskestrator/internal/opencode"
	"github.com/AntoineGS/taskestrator/internal/tracker"
	"github.com/AntoineGS/taskestrator/internal/ui"
)

const usage = `usage:
  tsk board            global board of dispatched tasks
  tsk project [dir]    one project's tasks, with dispatch (default: current directory)`

func main() {
	if err := run(os.Args[1:]); err != nil {
		fmt.Fprintln(os.Stderr, "tsk:", err)
		os.Exit(1)
	}
}

func run(args []string) error {
	if len(args) == 0 || (args[0] != "board" && args[0] != "project") {
		return errors.New(usage)
	}
	ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt)
	defer cancel()

	cfg, err := config.Load(config.DefaultPath())
	if err != nil {
		return err
	}
	if _, err := exec.LookPath("bd"); err != nil {
		return fmt.Errorf("bd is not on PATH: %w", err)
	}
	projects, warnings := discover.Scan(cfg.Roots, cfg.Hidden)
	bd := tracker.New()

	// Assign interfaces only on success, so they never hold a typed nil pointer.
	var live ui.Live
	oc, ocErr := opencode.FromService(ctx)
	if ocErr == nil {
		live = oc
	} else {
		warnings = append(warnings, ocErr.Error())
	}

	var h herdr.Herdr
	var jump ui.Jumper
	if _, err := exec.LookPath("herdr"); err == nil {
		cli := herdr.New()
		h = cli
		jump = func(ctx context.Context, t dispatch.Target) error { return dispatch.Jump(ctx, cli, t) }
	} else {
		warnings = append(warnings, "herdr is not on PATH; jumping and dispatch are disabled")
	}

	var model tea.Model
	switch args[0] {
	case "board":
		model = ui.NewBoard(ctx, ui.BoardConfig{
			Projects: projects, Tracker: bd, Live: live, Jump: jump,
			Reread: cfg.RereadInterval, Warnings: warnings,
		})
	case "project":
		dir := "."
		if len(args) > 1 {
			dir = args[1]
		}
		abs, err := filepath.Abs(dir)
		if err != nil {
			return err
		}
		p, ok := discover.ForDir(projects, abs)
		if !ok {
			return fmt.Errorf("no Beads project under %v contains %s; add its parent to roots in %s", cfg.Roots, abs, config.DefaultPath())
		}
		pf, err := config.LoadProjectFile(p.Repo)
		if err != nil {
			return err
		}
		var dispatcher ui.Dispatcher
		if ocErr == nil && h != nil {
			deps := dispatch.Deps{Tracker: bd, OpenCode: oc, Herdr: h, Git: gitwt.New(), Shell: execx.Command("sh")}
			dispatcher = func(ctx context.Context, req dispatch.Request) (dispatch.Result, error) {
				return dispatch.Run(ctx, deps, req)
			}
		}
		model = ui.NewProject(ctx, ui.ProjectConfig{
			Project: p, File: pf, Tracker: bd, Live: live, Jump: jump, Dispatch: dispatcher,
			Git: gitwt.New(), Reread: cfg.RereadInterval, Warnings: warnings,
		})
	}
	_, err = tea.NewProgram(model, tea.WithContext(ctx)).Run()
	return err
}

`README.md`:

````markdown
# tsk

Dispatch [Beads](https://github.com/gastownhall/beads) tasks to OpenCode agents
in [herdr](https://herdr.dev) tabs, and see the live state of every dispatched
task in one board.

## Requirements

- `bd`, with projects on the shared Dolt server
- the OpenCode v2 background service (`opencode service start`)
- `herdr`, for opening and jumping to tabs

## Install

```sh
go install github.com/AntoineGS/taskestrator/cmd/tsk@latest
```

## Use

- `tsk project [dir]` lists the project's open tasks, ready ones first. `d`
  dispatches the task under the cursor: it claims the task, creates
  `.worktrees/<id>` on branch `task/<id>`, starts an OpenCode session with the
  task as its prompt, and opens a herdr tab attached to that session. `enter`
  jumps to a dispatched task. `d` also re-dispatches a task whose session is
  lost.
- `tsk board` lists dispatched tasks from every project, grouped as waiting on
  you, in progress, review and lost. `enter` jumps to the task's tab, reopening
  it if it was closed. `a` adds queued tasks and tasks closed in the last day;
  `/` filters.

The agent finishes by running `bd update <id> --status review`. Review the
branch, merge it yourself, then run `bd close <id>`.

### Bind the board to a key in herdr

```toml
[[keys.command]]
type = "popup"
command = "tsk board"
```

### Per-project setup

Each project needs the `review` status once; `tsk project` offers to add it
with `R`. `.worktrees/` must be git-ignored.

An optional `.tsk.toml` in the repository:

```toml
setup = "make worktree-setup"   # run in each new worktree before the agent starts
prompt = """..."""              # Go template; fields: ID Title Description Acceptance Worktree Branch
```

### Configuration

`~/.config/tsk/config.toml`, every key optional:

```toml
roots = ["~/gits"]        # where to look for .beads/metadata.json
hidden = []               # project names or paths to skip
reread_interval = "10s"
```
````

func sh(t *testing.T, dir string, args ...string) string {
	t.Helper()
	cmd := exec.Command(args[0], args[1:]...)
	cmd.Dir = dir
	cmd.Env = append(os.Environ(), "GIT_AUTHOR_NAME=tsk", "GIT_AUTHOR_EMAIL=tsk@localhost",
		"GIT_COMMITTER_NAME=tsk", "GIT_COMMITTER_EMAIL=tsk@localhost", "BD_NON_INTERACTIVE=1")
	out, err := cmd.CombinedOutput()
	if err != nil {
		t.Fatalf("%v: %v: %s", args, err, out)
	}
	return strings.TrimSpace(string(out))
}

func TestDispatchEndToEnd(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	oc, err := opencode.FromService(ctx)
	if err != nil {
		t.Skipf("OpenCode service unavailable: %v", err)
	}
	if _, err := exec.LookPath("herdr"); err != nil {
		t.Skip("herdr is not on PATH")
	}

	repo := t.TempDir()
	sh(t, repo, "git", "init", "-q", "-b", "main")
	if err := os.WriteFile(filepath.Join(repo, ".gitignore"), []byte(".worktrees/\n"), 0o644); err != nil {
		t.Fatal(err)
	}
	sh(t, repo, "git", "add", ".gitignore")
	sh(t, repo, "git", "commit", "-q", "-m", "init")
	prefix := "tske2e" + time.Now().Format("150405")
	sh(t, repo, "bd", "init", "--prefix", prefix, "--skip-agents", "--skip-hooks", "--non-interactive")
	id := sh(t, repo, "bd", "create", "Say ok", "--silent")

	bd := tracker.New()
	if err := bd.AddReviewStatus(ctx, repo); err != nil {
		t.Fatal(err)
	}
	h := herdr.New()
	tasks, err := bd.Ready(ctx, repo)
	if err != nil || len(tasks) != 1 || tasks[0].ID != id {
		t.Fatalf("ready = %+v, err = %v", tasks, err)
	}

	res, err := dispatch.Run(ctx,
		dispatch.Deps{Tracker: bd, OpenCode: oc, Herdr: h, Git: gitwt.New(), Shell: execx.Command("sh")},
		dispatch.Request{
			Project: discover.Project{Name: prefix, Repo: repo},
			Task:    tasks[0],
			File:    config.ProjectFile{Prompt: "Reply with the single word ok. Do not run any tools."},
		})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		_ = oc.DeleteSession(context.Background(), res.SessionID)
		if ws, ok, _ := h.WorkspaceFor(context.Background(), repo); ok {
			_ = exec.Command("herdr", "workspace", "close", ws).Run()
		}
	})
	if res.TabError != nil {
		t.Fatalf("tab: %v", res.TabError)
	}

	list, err := bd.List(ctx, repo)
	if err != nil || len(list) != 1 || list[0].Meta(tracker.MetaSession) != res.SessionID || list[0].Status != "in_progress" {
		t.Fatalf("tasks = %+v, err = %v", list, err)
	}
	if b := sh(t, res.Worktree, "git", "rev-parse", "--abbrev-ref", "HEAD"); b != "task/"+id {
		t.Fatalf("branch = %q", b)
	}
	if _, err := oc.GetSession(ctx, res.SessionID); err != nil {
		t.Fatal(err)
	}
	deadline := time.Now().Add(20 * time.Second)
	for {
		if pane, ok, _ := h.PaneForSession(ctx, res.SessionID); ok && pane == res.PaneID {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("herdr never reported the session in the new pane")
		}
		time.Sleep(time.Second)
	}
}
